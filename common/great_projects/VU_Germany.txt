	start = 50
	date = 1884.01.01
	time = { months = 120 }
	build_cost = 1000
	starting_tier = 0
	type = monument
	can_be_moved = no
	build_trigger = {
		FROM = {
			OR = {
				tag = GER
				tag = HLR
			}
		}
        	culture_group = germanic
		OR = {
			has_owner_culture = yes
			has_owner_accepted_culture = yes
		}
	}
	on_built = { }
	on_destroyed = { }
	can_use_modifiers_trigger = {
		FROM = {
			OR = {
				tag = GER
				tag = PRU
				tag = HLR
				AND = {
					culture_group = germanic
					NOT = { primary_culture = dutch }
				}
			}
		}
		if = {
			limit = { has_dlc = "Common Sense" }
			FROM = {
				OR = {
					has_parliament = yes
					government = republic
				}
			}
		}
		else = {
			FROM = {
				OR = {
					government = republic
					has_reform = parliamentary_reform
				}
			}
		}
	}
	can_upgrade_trigger = {
		FROM = { adm_tech = 37 }
		FROM = {
			OR = {
				tag = GER
				tag = PRU
				tag = HLR
				AND = {
					culture_group = germanic
					NOT = { primary_culture = dutch }
				}
			}
		}
		if = {
			limit = { has_dlc = "Common Sense" }
			FROM = {
				OR = {
					has_parliament = yes
					government = republic
				}
			}
		}
		else = {
			FROM = {
				OR = {
					government = republic
					has_reform = parliamentary_reform
				}
			}
		}
	}
	keep_trigger = { }
	tier_0 = {
		upgrade_time = { months = 0 }
		cost_to_upgrade = { factor = 0 }
		province_modifiers = { }
		area_modifier = { }
		country_modifiers = { }
		on_upgraded = { }
	}
	tier_1 = {
		upgrade_time = { months = 120 }
		cost_to_upgrade = { factor = 1000 }
		province_modifiers = {
			local_monthly_devastation = -0.05
		}
		area_modifier = { }
		country_modifiers = { 
			governing_capacity = 0.1
		}
		on_upgraded = {
			add_devastation = -10
		}
	}
	tier_2 = {
		upgrade_time = { months = 240 }
		cost_to_upgrade = { factor = 2500 }
		province_modifiers = {
			local_monthly_devastation = -0.1
		}

		}
